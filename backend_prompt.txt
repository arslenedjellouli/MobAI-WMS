1. TECHNOLOGY STACK
===========================================

Core Backend:
- Django 4.2+
- Django REST Framework 3.14+
- PostgreSQL 14+ (with UUID, JSONB support)
- psycopg2-binary 2.9+

Authentication:
- djangorestframework-simplejwt 5.3+ (JWT tokens)
- django-cors-headers 4.3+ (CORS for mobile app)

Real-time:
- Django Channels 4.0+ (WebSocket support)
- channels-redis 4.1+ (WebSocket backend)
- daphne 4.0+ (ASGI server)

Async Tasks:
- Celery 5.3+
- Redis 4.6+ (message broker + cache)
- celery-beat 2.5+ (scheduled tasks)

AI Integration:
- httpx 0.25+ (async HTTP client for FastAPI)
- requests 2.31+ (sync HTTP client)

Utilities:
- Pillow 10.1+ (image processing)
- python-decouple 3.8+ (environment variables)
- django-filter 23.5+ (filtering support)
- drf-spectacular 0.27+ (API documentation)

Development:
- pytest-django 4.7+
- factory-boy 3.3+ (test data)
- django-debug-toolbar 4.2+
- black + flake8 (code formatting)

2. DATABASE SCHEMA (PostgreSQL)
===========================================

--- CORE TABLES ---

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(255) NOT NULL,
    role VARCHAR(20) NOT NULL,  -- 'ADMIN', 'SUPERVISOR', 'EMPLOYEE'
    status VARCHAR(20) DEFAULT 'ACTIVE',  -- 'ACTIVE', 'INACTIVE', 'SUSPENDED'
    phone VARCHAR(20),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMPTZ,
    CONSTRAINT users_role_check CHECK (role IN ('ADMIN', 'SUPERVISOR', 'EMPLOYEE')),
    CONSTRAINT users_status_check CHECK (status IN ('ACTIVE', 'INACTIVE', 'SUSPENDED'))
);

CREATE TABLE warehouses (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    address TEXT,
    city VARCHAR(100),
    country VARCHAR(100),
    postal_code VARCHAR(20),
    manager_id UUID REFERENCES users(id),
    status VARCHAR(20) DEFAULT 'ACTIVE',
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE zones (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    warehouse_id UUID NOT NULL REFERENCES warehouses(id) ON DELETE CASCADE,
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    zone_type VARCHAR(20) NOT NULL,  -- 'RECEIVING', 'VRAC', 'PICKING', 'EXPEDITION'
    capacity_max DECIMAL(10, 2),
    capacity_used DECIMAL(10, 2) DEFAULT 0,
    status VARCHAR(20) DEFAULT 'ACTIVE',
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT zones_type_check CHECK (zone_type IN ('RECEIVING', 'VRAC', 'PICKING', 'EXPEDITION'))
);

CREATE TABLE locations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    zone_id UUID NOT NULL REFERENCES zones(id) ON DELETE CASCADE,
    code VARCHAR(50) UNIQUE NOT NULL,
    aisle VARCHAR(20),
    rack VARCHAR(20),
    shelf VARCHAR(20),
    position VARCHAR(20),
    capacity DECIMAL(10, 2) NOT NULL,
    current_capacity_used DECIMAL(10, 2) DEFAULT 0,
    reserved_capacity DECIMAL(10, 2) DEFAULT 0,
    status VARCHAR(20) DEFAULT 'FREE',  -- 'FREE', 'ASSIGNED', 'OCCUPIED'
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT locations_status_check CHECK (status IN ('FREE', 'ASSIGNED', 'OCCUPIED')),
    CONSTRAINT locations_capacity_check CHECK (current_capacity_used + reserved_capacity <= capacity)
);

CREATE TABLE skus (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sku_code VARCHAR(100) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    barcode VARCHAR(100) UNIQUE,
    unit VARCHAR(20) DEFAULT 'UNIT',  -- 'UNIT', 'BOX', 'PALLET', 'KG', 'LITER'
    weight DECIMAL(10, 3),
    volume DECIMAL(10, 3),
    is_bulk BOOLEAN DEFAULT FALSE,
    min_picking_threshold DECIMAL(10, 2) DEFAULT 0,
    max_picking_capacity DECIMAL(10, 2) DEFAULT 0,
    category VARCHAR(100),
    image_url VARCHAR(500),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE stock (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sku_id UUID NOT NULL REFERENCES skus(id) ON DELETE CASCADE,
    location_id UUID NOT NULL REFERENCES locations(id) ON DELETE CASCADE,
    quantity DECIMAL(10, 2) NOT NULL DEFAULT 0,
    lot_number VARCHAR(100),
    expiration_date DATE,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT stock_quantity_check CHECK (quantity >= 0),
    CONSTRAINT stock_unique_sku_location UNIQUE (sku_id, location_id, lot_number)
);

CREATE TABLE stock_ledger (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sku_id UUID NOT NULL REFERENCES skus(id),
    location_id UUID REFERENCES locations(id),
    operation_type VARCHAR(30) NOT NULL,  -- 'RECEPTION', 'TRANSFER', 'PICKING', 'REPLENISHMENT', 'ADJUSTMENT', 'EXPEDITION'
    quantity_before DECIMAL(10, 2) NOT NULL,
    quantity DECIMAL(10, 2) NOT NULL,
    quantity_after DECIMAL(10, 2) NOT NULL,
    lot_number VARCHAR(100),
    reference_id UUID,
    reference_type VARCHAR(50),
    performed_by UUID REFERENCES users(id),
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT stock_ledger_operation_check CHECK (operation_type IN ('RECEPTION', 'TRANSFER', 'PICKING', 'REPLENISHMENT', 'ADJUSTMENT', 'EXPEDITION'))
);

CREATE TABLE chariots (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    code VARCHAR(50) UNIQUE NOT NULL,
    type VARCHAR(50),  -- 'PALLET_JACK', 'FORKLIFT', 'ORDER_PICKER', 'REACH_TRUCK'
    capacity DECIMAL(10, 2),
    status VARCHAR(20) DEFAULT 'AVAILABLE',  -- 'AVAILABLE', 'IN_USE', 'MAINTENANCE', 'OUT_OF_SERVICE'
    current_operator_id UUID REFERENCES users(id),
    last_maintenance_date DATE,
    next_maintenance_date DATE,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE daily_operations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    warehouse_id UUID NOT NULL REFERENCES warehouses(id),
    date DATE NOT NULL,
    shift VARCHAR(20),  -- 'MORNING', 'AFTERNOON', 'NIGHT'
    supervisor_id UUID REFERENCES users(id),
    notes TEXT,
    status VARCHAR(20) DEFAULT 'OPEN',  -- 'OPEN', 'CLOSED'
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT daily_operations_unique UNIQUE (warehouse_id, date, shift)
);

--- BON TABLES ---

CREATE TABLE bon_commande (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    numero VARCHAR(100) UNIQUE NOT NULL,
    fournisseur VARCHAR(255) NOT NULL,
    date_commande DATE NOT NULL,
    date_livraison_prevue DATE,
    warehouse_id UUID NOT NULL REFERENCES warehouses(id),
    status VARCHAR(30) DEFAULT 'EN_ATTENTE',  -- 'EN_ATTENTE', 'EN_RECEPTION', 'RECU', 'ANNULE'
    total_items INT DEFAULT 0,
    received_items INT DEFAULT 0,
    notes TEXT,
    created_by UUID REFERENCES users(id),
    received_by UUID REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_commande_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    bon_commande_id UUID NOT NULL REFERENCES bon_commande(id) ON DELETE CASCADE,
    sku_id UUID NOT NULL REFERENCES skus(id),
    quantity_ordered DECIMAL(10, 2) NOT NULL,
    quantity_received DECIMAL(10, 2) DEFAULT 0,
    lot_number VARCHAR(100),
    expiration_date DATE,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_preparation (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    numero VARCHAR(100) UNIQUE NOT NULL,
    client VARCHAR(255) NOT NULL,
    date_preparation DATE NOT NULL,
    warehouse_id UUID NOT NULL REFERENCES warehouses(id),
    status VARCHAR(30) DEFAULT 'EN_ATTENTE',  -- 'EN_ATTENTE', 'EN_PREPARATION', 'PRET', 'ANNULE'
    priority INT DEFAULT 0,
    total_items INT DEFAULT 0,
    prepared_items INT DEFAULT 0,
    assigned_to UUID REFERENCES users(id),
    notes TEXT,
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_preparation_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    bon_preparation_id UUID NOT NULL REFERENCES bon_preparation(id) ON DELETE CASCADE,
    sku_id UUID NOT NULL REFERENCES skus(id),
    quantity_requested DECIMAL(10, 2) NOT NULL,
    quantity_prepared DECIMAL(10, 2) DEFAULT 0,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_replenishment (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    numero VARCHAR(100) UNIQUE NOT NULL,
    sku_id UUID NOT NULL REFERENCES skus(id),
    from_location_id UUID NOT NULL REFERENCES locations(id),  -- VRAC zone
    to_location_id UUID NOT NULL REFERENCES locations(id),    -- PICKING zone
    quantity DECIMAL(10, 2) NOT NULL,
    status VARCHAR(30) DEFAULT 'EN_ATTENTE',  -- 'EN_ATTENTE', 'EN_REPLENISSEMENT', 'COMPLETE', 'ANNULE'
    trigger_type VARCHAR(30),  -- 'MANUAL', 'AUTOMATIC', 'AI_SUGGESTED'
    priority INT DEFAULT 0,
    assigned_to UUID REFERENCES users(id),
    completed_by UUID REFERENCES users(id),
    completed_at TIMESTAMPTZ,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_replenishment_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    bon_replenishment_id UUID NOT NULL REFERENCES bon_replenishment(id) ON DELETE CASCADE,
    lot_number VARCHAR(100),
    quantity DECIMAL(10, 2) NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_picking (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    numero VARCHAR(100) UNIQUE NOT NULL,
    bon_preparation_id UUID REFERENCES bon_preparation(id),
    warehouse_id UUID NOT NULL REFERENCES warehouses(id),
    status VARCHAR(30) DEFAULT 'EN_ATTENTE',  -- 'EN_ATTENTE', 'EN_PICKING', 'COMPLETE', 'ANNULE'
    priority INT DEFAULT 0,
    assigned_to UUID REFERENCES users(id),
    chariot_id UUID REFERENCES chariots(id),
    picking_path JSONB,  -- AI-generated optimal path stored as JSON
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_picking_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    bon_picking_id UUID NOT NULL REFERENCES bon_picking(id) ON DELETE CASCADE,
    sku_id UUID NOT NULL REFERENCES skus(id),
    location_id UUID NOT NULL REFERENCES locations(id),
    quantity_requested DECIMAL(10, 2) NOT NULL,
    quantity_picked DECIMAL(10, 2) DEFAULT 0,
    lot_number VARCHAR(100),
    expiration_date DATE,
    sequence INT,  -- Order in picking path
    picked_at TIMESTAMPTZ,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_livraison (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    numero VARCHAR(100) UNIQUE NOT NULL,
    bon_preparation_id UUID REFERENCES bon_preparation(id),
    client VARCHAR(255) NOT NULL,
    adresse_livraison TEXT NOT NULL,
    date_livraison DATE,
    warehouse_id UUID NOT NULL REFERENCES warehouses(id),
    status VARCHAR(30) DEFAULT 'EN_ATTENTE',  -- 'EN_ATTENTE', 'EN_LIVRAISON', 'LIVRE', 'ANNULE'
    transporteur VARCHAR(255),
    numero_suivi VARCHAR(100),
    assigned_driver_id UUID REFERENCES users(id),
    chariot_id UUID REFERENCES chariots(id),
    delivered_at TIMESTAMPTZ,
    signature_url VARCHAR(500),
    notes TEXT,
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE bon_livraison_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    bon_livraison_id UUID NOT NULL REFERENCES bon_livraison(id) ON DELETE CASCADE,
    sku_id UUID NOT NULL REFERENCES skus(id),
    quantity DECIMAL(10, 2) NOT NULL,
    lot_number VARCHAR(100),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

--- RESERVATION & AI TABLES ---

CREATE TABLE location_reservations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    location_id UUID NOT NULL REFERENCES locations(id) ON DELETE CASCADE,
    reserved_for VARCHAR(50) NOT NULL,  -- 'BON_COMMANDE', 'BON_REPLENISHMENT'
    reference_id UUID NOT NULL,
    reserved_capacity DECIMAL(10, 2) NOT NULL,
    reserved_by UUID REFERENCES users(id),
    status VARCHAR(20) DEFAULT 'ACTIVE',  -- 'ACTIVE', 'COMPLETED', 'CANCELLED'
    reserved_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    CONSTRAINT location_reservations_status_check CHECK (status IN ('ACTIVE', 'COMPLETED', 'CANCELLED'))
);

CREATE TABLE picking_stock_status (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sku_id UUID NOT NULL REFERENCES skus(id) ON DELETE CASCADE,
    zone_id UUID NOT NULL REFERENCES zones(id) ON DELETE CASCADE,
    current_quantity DECIMAL(10, 2) NOT NULL DEFAULT 0,
    min_threshold DECIMAL(10, 2) NOT NULL,
    max_capacity DECIMAL(10, 2) NOT NULL,
    status VARCHAR(20) DEFAULT 'OK',  -- 'OK', 'LOW', 'CRITICAL', 'EMPTY'
    last_checked_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT picking_stock_unique UNIQUE (sku_id, zone_id)
);

CREATE TABLE ai_decisions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    decision_type VARCHAR(50) NOT NULL,  -- 'PICKING_PATH', 'STOCK_PREDICTION', 'REPLENISHMENT', 'LOCATION_ASSIGNMENT'
    reference_type VARCHAR(50),
    reference_id UUID,
    input_data JSONB NOT NULL,
    output_data JSONB,
    confidence_score DECIMAL(5, 4),
    status VARCHAR(20) DEFAULT 'PENDING',  -- 'PENDING', 'COMPLETED', 'FAILED', 'APPLIED'
    requested_by UUID REFERENCES users(id),
    applied_by UUID REFERENCES users(id),
    requested_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMPTZ,
    applied_at TIMESTAMPTZ,
    error_message TEXT
);

--- OFFLINE SYNC TABLES ---

CREATE TABLE sync_queue (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id),
    entity_type VARCHAR(50) NOT NULL,  -- 'BON_PICKING', 'STOCK_MOVEMENT', etc.
    entity_id UUID,
    operation VARCHAR(20) NOT NULL,  -- 'CREATE', 'UPDATE', 'DELETE'
    data JSONB NOT NULL,
    status VARCHAR(20) DEFAULT 'PENDING',  -- 'PENDING', 'PROCESSING', 'COMPLETED', 'CONFLICT', 'FAILED'
    conflict_data JSONB,
    resolved_by UUID REFERENCES users(id),
    priority INT DEFAULT 0,
    client_timestamp TIMESTAMPTZ NOT NULL,
    server_timestamp TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    processed_at TIMESTAMPTZ,
    error_message TEXT
);

CREATE TABLE sync_conflicts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sync_queue_id UUID NOT NULL REFERENCES sync_queue(id) ON DELETE CASCADE,
    employee_id UUID NOT NULL REFERENCES users(id),
    supervisor_id UUID REFERENCES users(id),
    entity_type VARCHAR(50) NOT NULL,
    entity_id UUID,
    client_data JSONB NOT NULL,
    server_data JSONB NOT NULL,
    status VARCHAR(20) DEFAULT 'PENDING',  -- 'PENDING', 'APPROVED', 'REJECTED', 'MERGED'
    resolution_notes TEXT,
    resolved_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

--- AUDIT & NOTIFICATIONS ---

CREATE TABLE audit_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    action VARCHAR(100) NOT NULL,
    entity_type VARCHAR(50),
    entity_id UUID,
    ip_address INET,
    user_agent TEXT,
    changes JSONB,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE notifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id),
    type VARCHAR(50) NOT NULL,  -- 'SYNC_CONFLICT', 'LOW_STOCK', 'BON_ASSIGNED', 'SYSTEM_ALERT'
    title VARCHAR(255) NOT NULL,
    message TEXT NOT NULL,
    data JSONB,
    status VARCHAR(20) DEFAULT 'UNREAD',  -- 'UNREAD', 'READ', 'DISMISSED'
    priority INT DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    read_at TIMESTAMPTZ
);

8. OFFLINE SYNC SYSTEM
===========================================

--- Sync Flow ---

1. Mobile app collects offline operations in local SQLite
2. When online, app sends batch to POST /sync/upload
3. Django creates SyncQueue entries
4. Celery task processes queue sequentially
5. If conflict detected:
   - Create SyncConflict entry
   - Notify employee and supervisor via WebSocket
   - Supervisor resolves conflict via POST /sync/conflicts/{id}/resolve
6. Update mobile app via WebSocket

--- Conflict Detection Logic ---

Conflicts occur when:
1. Entity was modified on server after client's last sync
2. Entity was deleted on server
3. Business rule violation (e.g., insufficient stock)

Conflict Resolution Strategy:
- Employee notifies server of offline change
- Server detects conflict and creates SyncConflict
- Server notifies supervisor via WebSocket + Notification
- Supervisor reviews both versions (client vs server data)
- Supervisor approves, rejects, or merges
- Server applies resolution
- Server notifies employee via WebSocket

--- Sync Queue Processing ---

@shared_task
def process_sync_queue():
    """Process pending sync queue items"""
    queue_items = SyncQueue.objects.filter(status='PENDING').order_by('client_timestamp', 'priority')
    
    for item in queue_items:
        try:
            item.status = 'PROCESSING'
            item.save()
            
            # Apply operation
            if item.operation == 'CREATE':
                create_entity(item)
            elif item.operation == 'UPDATE':
                update_entity(item)
            elif item.operation == 'DELETE':
                delete_entity(item)
            
            item.status = 'COMPLETED'
            item.processed_at = timezone.now()
            item.save()
            
        except ConflictException as e:
            # Create conflict for supervisor
            conflict = SyncConflict.objects.create(
                sync_queue_id=item.id,
                employee_id=item.user_id,
                entity_type=item.entity_type,
                entity_id=item.entity_id,
                client_data=item.data,
                server_data=get_current_server_data(item),
                status='PENDING'
            )
            
            # Notify supervisor
            notify_supervisor_of_conflict(conflict)
            
            item.status = 'CONFLICT'
            item.conflict_data = e.details
            item.save()
            
        except Exception as e:
            item.status = 'FAILED'
            item.error_message = str(e)
            item.save()
