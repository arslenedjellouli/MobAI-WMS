================================================================================
MOBAI WAREHOUSE MANAGEMENT SYSTEM - COMPLETE AI AGENT PROMPT
================================================================================

PROJECT OVERVIEW:
You are building an enterprise-grade mobile Warehouse Management System (WMS) 
with AI-powered optimization for the MobAI hackathon. This system manages the 
complete warehouse operation flow: Receipt → Transfer → Preparation → Picking → Delivery.

The system has 3 roles: ADMIN, SUPERVISOR, EMPLOYEE
- AI generates suggestions for storage locations and picking routes
- SUPERVISOR and ADMIN can override AI decisions with justification
- EMPLOYEE only sees validated final orders (abstraction layer)

================================================================================
PART 1: COMPLETE DATABASE SCHEMA
================================================================================

--- 0. DAILY OPERATION CONTEXT ---

CREATE TABLE DAILY_OPERATION (
    id INT PRIMARY KEY AUTO_INCREMENT,
    operation_date DATE UNIQUE NOT NULL,
    total_chariots_available INT NOT NULL DEFAULT 10,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_operation_date (operation_date)
) ENGINE=InnoDB;

--- 1. AUTHENTICATION & SECURITY ---

CREATE TABLE USERS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('ADMIN', 'SUPERVISOR', 'EMPLOYEE') NOT NULL,
    status ENUM('ACTIVE', 'DISABLED') DEFAULT 'ACTIVE',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_email (email),
    INDEX idx_role (role)
) ENGINE=InnoDB;

CREATE TABLE USER_SESSIONS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    login_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    logout_time TIMESTAMP NULL,
    device_info VARCHAR(255),
    ip_address VARCHAR(45),
    FOREIGN KEY (user_id) REFERENCES USERS(id) ON DELETE CASCADE,
    INDEX idx_user_session (user_id, login_time)
) ENGINE=InnoDB;

CREATE TABLE ACTION_LOGS (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    action_type VARCHAR(50) NOT NULL,
    entity_type VARCHAR(50) NOT NULL,
    entity_id INT,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    metadata JSON,
    FOREIGN KEY (user_id) REFERENCES USERS(id),
    INDEX idx_action_timestamp (timestamp),
    INDEX idx_action_user (user_id),
    INDEX idx_action_entity (entity_type, entity_id)
) ENGINE=InnoDB;

--- 2. WAREHOUSE STRUCTURE ---

CREATE TABLE WAREHOUSES (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) UNIQUE NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_code (code)
) ENGINE=InnoDB;

CREATE TABLE FLOORS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    warehouse_id INT NOT NULL,
    level_number INT NOT NULL CHECK (level_number >= 0 AND level_number <= 4),
    floor_type ENUM('GROUND', 'STORAGE') NOT NULL,
    description VARCHAR(255),
    matrix_rows INT UNSIGNED DEFAULT 10,
    matrix_columns INT UNSIGNED DEFAULT 10,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (warehouse_id) REFERENCES WAREHOUSES(id) ON DELETE CASCADE,
    UNIQUE KEY unique_floor_level (warehouse_id, level_number),
    INDEX idx_warehouse_floor (warehouse_id, level_number)
) ENGINE=InnoDB;

CREATE TABLE SECTIONS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    floor_id INT NOT NULL,
    section_type ENUM('PICKING', 'ELEVATOR', 'MONTE_CHARGE_1', 'MONTE_CHARGE_2', 
                      'VRAC', 'EXPEDITION_1', 'EXPEDITION_2', 'BUREAU', 
                      'PALETTE_SLOTS', 'ROUTE_SLOTS') NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    start_row INT UNSIGNED NOT NULL,
    end_row INT UNSIGNED NOT NULL,
    start_column INT UNSIGNED NOT NULL,
    end_column INT UNSIGNED NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (floor_id) REFERENCES FLOORS(id) ON DELETE CASCADE,
    UNIQUE KEY unique_floor_section_type (floor_id, section_type),
    INDEX idx_floor_section (floor_id, section_type)
) ENGINE=InnoDB;

CREATE TABLE SLOTS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    floor_id INT NOT NULL,
    section_id INT NOT NULL,
    row INT UNSIGNED NOT NULL,
    `column` INT UNSIGNED NOT NULL,
    code VARCHAR(50) UNIQUE NOT NULL,
    is_storage BOOLEAN DEFAULT FALSE,
    is_infrastructure BOOLEAN DEFAULT FALSE,
    is_aisle BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (floor_id) REFERENCES FLOORS(id) ON DELETE CASCADE,
    FOREIGN KEY (section_id) REFERENCES SECTIONS(id) ON DELETE CASCADE,
    UNIQUE KEY unique_floor_slot_position (floor_id, row, `column`),
    INDEX idx_floor_slot (floor_id, row, `column`),
    INDEX idx_section_slot (section_id)
) ENGINE=InnoDB;

CREATE TABLE ZONES (
    id INT PRIMARY KEY AUTO_INCREMENT,
    warehouse_id INT NOT NULL,
    name VARCHAR(255) NOT NULL,
    type ENUM('RECEIVING', 'VRAC', 'PICKING', 'EXPEDITION') NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (warehouse_id) REFERENCES WAREHOUSES(id) ON DELETE CASCADE,
    INDEX idx_zone_warehouse (warehouse_id),
    INDEX idx_zone_type (type)
) ENGINE=InnoDB;

CREATE TABLE LOCATIONS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    warehouse_id INT NOT NULL,
    floor_id INT NULL,
    zone_id INT NOT NULL,
    slot_id INT NULL,
    code VARCHAR(50) UNIQUE NOT NULL,
    aisle VARCHAR(20),
    rack VARCHAR(20),
    shelf VARCHAR(20),
    position VARCHAR(20),
    x_coord DECIMAL(10, 2),
    y_coord DECIMAL(10, 2),
    capacity DECIMAL(10, 2) NOT NULL DEFAULT 1.3,
    current_capacity_used DECIMAL(10, 2) DEFAULT 0,
    reserved_capacity DECIMAL(10, 2) DEFAULT 0,
    status ENUM('FREE', 'ASSIGNED', 'OCCUPIED') DEFAULT 'FREE',
    is_active BOOLEAN DEFAULT TRUE,
    is_pallet_position BOOLEAN DEFAULT TRUE,
    max_pallets INT DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (warehouse_id) REFERENCES WAREHOUSES(id) ON DELETE CASCADE,
    FOREIGN KEY (floor_id) REFERENCES FLOORS(id) ON DELETE SET NULL,
    FOREIGN KEY (zone_id) REFERENCES ZONES(id) ON DELETE CASCADE,
    FOREIGN KEY (slot_id) REFERENCES SLOTS(id) ON DELETE SET NULL,
    CONSTRAINT check_capacity_valid CHECK (current_capacity_used + reserved_capacity <= capacity),
    CONSTRAINT check_capacity_positive CHECK (capacity >= 0),
    CONSTRAINT check_used_positive CHECK (current_capacity_used >= 0),
    CONSTRAINT check_reserved_positive CHECK (reserved_capacity >= 0),
    INDEX idx_location_code (code),
    INDEX idx_location_zone (zone_id),
    INDEX idx_location_slot (slot_id),
    INDEX idx_location_coords (x_coord, y_coord),
    INDEX idx_location_active (is_active),
    INDEX idx_location_status (status)
) ENGINE=InnoDB;

--- 3. SKU & INVENTORY ---

CREATE TABLE SKUS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    sku_code VARCHAR(100) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    weight DECIMAL(10, 2),
    min_picking_threshold INT DEFAULT 0,
    max_picking_capacity INT NULL,
    is_bulk BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_sku_code (sku_code),
    INDEX idx_sku_bulk (is_bulk)
) ENGINE=InnoDB;

CREATE TABLE SKU_STATISTICS (
    sku_id INT PRIMARY KEY,
    demand_frequency DECIMAL(10, 4) DEFAULT 0,
    delivery_frequency DECIMAL(10, 4) DEFAULT 0,
    reception_frequency DECIMAL(10, 4) DEFAULT 0,
    avg_daily_demand DECIMAL(10, 2) DEFAULT 0,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (sku_id) REFERENCES SKUS(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE STOCK (
    sku_id INT NOT NULL,
    location_id INT NOT NULL,
    quantity INT NOT NULL DEFAULT 0,
    version INT NOT NULL DEFAULT 1,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (sku_id, location_id),
    FOREIGN KEY (sku_id) REFERENCES SKUS(id) ON DELETE CASCADE,
    FOREIGN KEY (location_id) REFERENCES LOCATIONS(id) ON DELETE CASCADE,
    CONSTRAINT check_positive_quantity CHECK (quantity >= 0),
    CONSTRAINT check_version_positive CHECK (version >= 1),
    INDEX idx_stock_location (location_id),
    INDEX idx_stock_quantity (quantity)
) ENGINE=InnoDB;

CREATE TABLE STOCK_LEDGER (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    sku_id INT NOT NULL,
    from_location_id INT NULL,
    to_location_id INT NULL,
    quantity INT NOT NULL,
    operation_type ENUM('RECEIPT', 'TRANSFER', 'REPLENISHMENT', 'PICKING', 'DELIVERY') NOT NULL,
    performed_by INT NOT NULL,
    validated_by INT NULL,
    override_flag BOOLEAN DEFAULT FALSE,
    justification TEXT NULL,
    reference_document_type ENUM('BON_COMMANDE', 'BON_PREPARATION', 'BON_REPLENISHMENT', 'BON_PICKING', 'BON_LIVRAISON') NULL,
    reference_document_id INT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sku_id) REFERENCES SKUS(id),
    FOREIGN KEY (from_location_id) REFERENCES LOCATIONS(id),
    FOREIGN KEY (to_location_id) REFERENCES LOCATIONS(id),
    FOREIGN KEY (performed_by) REFERENCES USERS(id),
    FOREIGN KEY (validated_by) REFERENCES USERS(id),
    INDEX idx_ledger_timestamp (timestamp),
    INDEX idx_ledger_sku (sku_id),
    INDEX idx_ledger_operation (operation_type),
    INDEX idx_ledger_reference (reference_document_type, reference_document_id)
) ENGINE=InnoDB;

CREATE TABLE LOCATION_RESERVATIONS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    location_id INT NOT NULL,
    sku_id INT NOT NULL,
    quantity INT NOT NULL,
    reservation_type ENUM('RECEPTION', 'REPLENISHMENT', 'TRANSFER') NOT NULL,
    reference_document_type ENUM('BON_COMMANDE', 'BON_PREPARATION', 'BON_REPLENISHMENT', 'BON_PICKING') NOT NULL,
    reference_document_id INT NOT NULL,
    status ENUM('ACTIVE', 'COMPLETED', 'CANCELLED') DEFAULT 'ACTIVE',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP NULL,
    FOREIGN KEY (location_id) REFERENCES LOCATIONS(id) ON DELETE CASCADE,
    FOREIGN KEY (sku_id) REFERENCES SKUS(id),
    CONSTRAINT check_reservation_quantity_positive CHECK (quantity > 0),
    INDEX idx_reservation_location (location_id),
    INDEX idx_reservation_sku (sku_id),
    INDEX idx_reservation_status (status),
    INDEX idx_reservation_reference (reference_document_type, reference_document_id)
) ENGINE=InnoDB;

CREATE TABLE PICKING_STOCK_STATUS (
    sku_id INT NOT NULL,
    location_id INT NOT NULL,
    current_quantity INT NOT NULL DEFAULT 0,
    last_checked_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (sku_id, location_id),
    FOREIGN KEY (sku_id) REFERENCES SKUS(id) ON DELETE CASCADE,
    FOREIGN KEY (location_id) REFERENCES LOCATIONS(id) ON DELETE CASCADE,
    INDEX idx_picking_stock_location (location_id),
    INDEX idx_picking_stock_quantity (current_quantity)
) ENGINE=InnoDB;

    assigned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (chariot_id) REFERENCES CHARIOTS(id) ON DELETE CASCADE,

--- 5. ORDER SYSTEM (BON-BASED WORKFLOW) ---

-- BON DE COMMANDE (Incoming Products)
CREATE TABLE BON_COMMANDE (
    id INT PRIMARY KEY AUTO_INCREMENT,
    daily_operation_id INT NOT NULL,
    created_by INT NOT NULL,
    status ENUM('PENDING', 'RECEIVED', 'CLOSED') DEFAULT 'PENDING',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (daily_operation_id) REFERENCES DAILY_OPERATION(id) ON DELETE CASCADE,
    FOREIGN KEY (created_by) REFERENCES USERS(id),
    INDEX idx_bon_commande_daily (daily_operation_id),
    INDEX idx_bon_commande_status (status),
    INDEX idx_bon_commande_created (created_at)
) ENGINE=InnoDB;

CREATE TABLE BON_COMMANDE_ITEMS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    bon_commande_id INT NOT NULL,
    sku_id INT NOT NULL,
    quantity_received INT NOT NULL,
    reception_datetime TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (bon_commande_id) REFERENCES BON_COMMANDE(id) ON DELETE CASCADE,
    FOREIGN KEY (sku_id) REFERENCES SKUS(id),
    INDEX idx_bon_commande_item_order (bon_commande_id)
) ENGINE=InnoDB;

-- BON DE PREPARATION (Outgoing Planning - AI Generated)
CREATE TABLE BON_PREPARATION (
    delivery_id INT PRIMARY KEY AUTO_INCREMENT,
    daily_operation_id INT NOT NULL,
    status ENUM('AI_GENERATED', 'VALIDATED', 'OVERRIDDEN', 'EXECUTED') DEFAULT 'AI_GENERATED',
    generated_by_ai BOOLEAN DEFAULT FALSE,
    validated_by INT NULL,
    confidence_score DECIMAL(5, 4) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (daily_operation_id) REFERENCES DAILY_OPERATION(id) ON DELETE CASCADE,
    FOREIGN KEY (validated_by) REFERENCES USERS(id),
    INDEX idx_bon_prep_daily (daily_operation_id),
    INDEX idx_bon_prep_status (status),
    INDEX idx_bon_prep_created (created_at)
) ENGINE=InnoDB;

CREATE TABLE BON_PREPARATION_ITEMS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    delivery_id INT NOT NULL,
    sku_id INT NOT NULL,
    quantity_to_deliver INT NOT NULL,
    current_storage_location_id INT NOT NULL,
    FOREIGN KEY (delivery_id) REFERENCES BON_PREPARATION(delivery_id) ON DELETE CASCADE,
    FOREIGN KEY (sku_id) REFERENCES SKUS(id),
    FOREIGN KEY (current_storage_location_id) REFERENCES LOCATIONS(id),
    INDEX idx_bon_prep_item_delivery (delivery_id)
) ENGINE=InnoDB;

-- BON DE PICKING (Optimized Route Execution)
CREATE TABLE BON_PICKING (
    id INT PRIMARY KEY AUTO_INCREMENT,
    delivery_id INT NOT NULL,
    status ENUM('PENDING', 'IN_PROGRESS', 'COMPLETED') DEFAULT 'PENDING',
    assigned_chariot_id INT NULL,
    total_distance DECIMAL(10, 2) NULL,
    optimized BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (delivery_id) REFERENCES BON_PREPARATION(delivery_id) ON DELETE CASCADE,
    FOREIGN KEY (assigned_chariot_id) REFERENCES CHARIOTS(id) ON DELETE SET NULL,
    INDEX idx_bon_picking_delivery (delivery_id),
    INDEX idx_bon_picking_status (status),
    INDEX idx_bon_picking_created (created_at)
) ENGINE=InnoDB;

CREATE TABLE BON_PICKING_ITEMS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    bon_picking_id INT NOT NULL,
    sku_id INT NOT NULL,
    quantity INT NOT NULL,
    source_storage_location_id INT NOT NULL,
    destination_picking_location_id INT NOT NULL,
    sequence_number INT NOT NULL,
    FOREIGN KEY (bon_picking_id) REFERENCES BON_PICKING(id) ON DELETE CASCADE,
    FOREIGN KEY (sku_id) REFERENCES SKUS(id),
    FOREIGN KEY (source_storage_location_id) REFERENCES LOCATIONS(id),
    FOREIGN KEY (destination_picking_location_id) REFERENCES LOCATIONS(id),
    INDEX idx_bon_picking_item_order (bon_picking_id),
    INDEX idx_bon_picking_sequence (bon_picking_id, sequence_number)
) ENGINE=InnoDB;

-- BON DE LIVRAISON (Delivery Validation)
CREATE TABLE BON_LIVRAISON (
    id INT PRIMARY KEY AUTO_INCREMENT,
    delivery_id INT NOT NULL,
    status ENUM('IDLE', 'IN_PROGRESS', 'DONE', 'FAILED') DEFAULT 'IDLE',
    validated_by INT NULL,
    notes TEXT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (delivery_id) REFERENCES BON_PREPARATION(delivery_id) ON DELETE CASCADE,
    FOREIGN KEY (validated_by) REFERENCES USERS(id),
    INDEX idx_bon_livraison_delivery (delivery_id),
    INDEX idx_bon_livraison_status (status),
    INDEX idx_bon_livraison_updated (updated_at)
) ENGINE=InnoDB;

-- BON DE REPLENISHMENT (VRAC → PICKING Transfers)
CREATE TABLE BON_REPLENISHMENT (
    id INT PRIMARY KEY AUTO_INCREMENT,
    daily_operation_id INT NOT NULL,
    status ENUM('PENDING', 'IN_PROGRESS', 'DONE', 'FAILED') DEFAULT 'PENDING',
    generated_by_ai BOOLEAN DEFAULT FALSE,
    validated_by INT NULL,
    assigned_chariot_id INT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (daily_operation_id) REFERENCES DAILY_OPERATION(id) ON DELETE CASCADE,
    FOREIGN KEY (validated_by) REFERENCES USERS(id),
    FOREIGN KEY (assigned_chariot_id) REFERENCES CHARIOTS(id) ON DELETE SET NULL,
    INDEX idx_bon_replenishment_daily (daily_operation_id),
    INDEX idx_bon_replenishment_status (status),
    INDEX idx_bon_replenishment_created (created_at)
) ENGINE=InnoDB;

CREATE TABLE BON_REPLENISHMENT_ITEMS (
    id INT PRIMARY KEY AUTO_INCREMENT,
    bon_replenishment_id INT NOT NULL,
    sku_id INT NOT NULL,
    quantity INT NOT NULL,
    from_vrac_location_id INT NOT NULL,
    to_picking_location_id INT NOT NULL,
    sequence_number INT NOT NULL,
    FOREIGN KEY (bon_replenishment_id) REFERENCES BON_REPLENISHMENT(id) ON DELETE CASCADE,
    FOREIGN KEY (sku_id) REFERENCES SKUS(id),
    FOREIGN KEY (from_vrac_location_id) REFERENCES LOCATIONS(id),
    FOREIGN KEY (to_picking_location_id) REFERENCES LOCATIONS(id),
    CONSTRAINT check_replenishment_quantity_positive CHECK (quantity > 0),
    INDEX idx_bon_replenishment_item_order (bon_replenishment_id),
    INDEX idx_bon_replenishment_sequence (bon_replenishment_id, sequence_number)
) ENGINE=InnoDB;

    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    device_id INT NOT NULL,
    operation_payload JSON NOT NULL,
    status ENUM('PENDING', 'PROCESSED', 'FAILED') DEFAULT 'PENDING',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (device_id) REFERENCES DEVICES(id) ON DELETE CASCADE,
    INDEX idx_sync_status (status),
    INDEX idx_sync_device (device_id, status),
    INDEX idx_sync_created (created_at)
) ENGINE=InnoDB;

--- TRIGGER: Update STOCK, LOCATION_TRAFFIC, and LOCATION status from STOCK_LEDGER ---
DELIMITER //
CREATE TRIGGER after_ledger_insert 
AFTER INSERT ON STOCK_LEDGER
FOR EACH ROW
BEGIN
    -- Update stock at destination location
    IF NEW.to_location_id IS NOT NULL THEN
        INSERT INTO STOCK (sku_id, location_id, quantity, version)
        VALUES (NEW.sku_id, NEW.to_location_id, NEW.quantity, 1)
        ON DUPLICATE KEY UPDATE 
            quantity = quantity + NEW.quantity,
            version = version + 1;
        
        -- Update location capacity and status
        UPDATE LOCATIONS
        SET current_capacity_used = current_capacity_used + NEW.quantity,
            status = CASE
                WHEN current_capacity_used + NEW.quantity > 0 THEN 'OCCUPIED'
                WHEN reserved_capacity > 0 THEN 'ASSIGNED'
                ELSE 'FREE'
            END
        WHERE id = NEW.to_location_id;
    END IF;
    
    -- Update stock at source location
    IF NEW.from_location_id IS NOT NULL THEN
        UPDATE STOCK 
        SET quantity = quantity - NEW.quantity,
            version = version + 1
        WHERE sku_id = NEW.sku_id 
        AND location_id = NEW.from_location_id;
        
        -- Update location capacity and status
        UPDATE LOCATIONS
        SET current_capacity_used = GREATEST(0, current_capacity_used - NEW.quantity),
            status = CASE
                WHEN current_capacity_used - NEW.quantity <= 0 AND reserved_capacity > 0 THEN 'ASSIGNED'
                WHEN current_capacity_used - NEW.quantity <= 0 AND reserved_capacity <= 0 THEN 'FREE'
                ELSE 'OCCUPIED'
            END
        WHERE id = NEW.from_location_id;
    END IF;
    
    -- Update global traffic statistics
    IF NEW.to_location_id IS NOT NULL THEN
        INSERT INTO LOCATION_TRAFFIC (location_id, movement_count)
        VALUES (NEW.to_location_id, 1)
        ON DUPLICATE KEY UPDATE 
            movement_count = movement_count + 1;
    END IF;
    
    IF NEW.from_location_id IS NOT NULL THEN
        INSERT INTO LOCATION_TRAFFIC (location_id, movement_count)
        VALUES (NEW.from_location_id, 1)
        ON DUPLICATE KEY UPDATE 
            movement_count = movement_count + 1;
    END IF;
    
    -- Update PICKING_STOCK_STATUS if operation involves picking zone
    IF NEW.operation_type IN ('REPLENISHMENT', 'PICKING') THEN
        IF NEW.to_location_id IS NOT NULL THEN
            INSERT INTO PICKING_STOCK_STATUS (sku_id, location_id, current_quantity)
            SELECT NEW.sku_id, NEW.to_location_id, NEW.quantity
            ON DUPLICATE KEY UPDATE
                current_quantity = current_quantity + NEW.quantity;
        END IF;
        
        IF NEW.from_location_id IS NOT NULL THEN
            UPDATE PICKING_STOCK_STATUS
            SET current_quantity = GREATEST(0, current_quantity - NEW.quantity)
            WHERE sku_id = NEW.sku_id AND location_id = NEW.from_location_id;
        END IF;
    END IF;
END//
DELIMITER ;

CREATE TRIGGER after_reservation_insert
AFTER INSERT ON LOCATION_RESERVATIONS
FOR EACH ROW
BEGIN

        UPDATE LOCATIONS
        SET reserved_capacity = reserved_capacity + NEW.quantity,
            status = CASE
                WHEN current_capacity_used > 0 THEN 'OCCUPIED'
                WHEN reserved_capacity + NEW.quantity > 0 THEN 'ASSIGNED'

            END
        WHERE id = NEW.location_id;

END//

--- TRIGGER: Update LOCATION reserved_capacity when reservation updated ---
DELIMITER //
CREATE TRIGGER after_reservation_update
AFTER UPDATE ON LOCATION_RESERVATIONS
FOR EACH ROW
BEGIN
    -- If reservation was completed or cancelled, reduce reserved capacity
    IF OLD.status = 'ACTIVE' AND NEW.status IN ('COMPLETED', 'CANCELLED') THEN
        UPDATE LOCATIONS
        SET reserved_capacity = GREATEST(0, reserved_capacity - OLD.quantity),
            status = CASE
                WHEN current_capacity_used > 0 THEN 'OCCUPIED'
                WHEN reserved_capacity - OLD.quantity > 0 THEN 'ASSIGNED'
                ELSE 'FREE'
            END
        WHERE id = OLD.location_id;
    END IF;
END//

--- PROCEDURE: Calculate total stock for SKU ---
DELIMITER //
CREATE PROCEDURE get_total_stock(IN p_sku_id INT)
BEGIN
    SELECT 
        s.sku_code,
        s.name,
        SUM(st.quantity) as total_quantity,
        COUNT(DISTINCT st.location_id) as location_count
    FROM SKUS s
    LEFT JOIN STOCK st ON s.id = st.sku_id
    WHERE s.id = p_sku_id
    GROUP BY s.id, s.sku_code, s.name;
END//

--- PROCEDURE: Check available capacity at location ---
DELIMITER //
CREATE PROCEDURE check_available_capacity(
    IN p_location_id INT
)
BEGIN
    SELECT 
        l.id,
        l.code,
        l.capacity,
        l.current_capacity_used,
        l.reserved_capacity,
        (l.capacity - l.current_capacity_used - l.reserved_capacity) as available_capacity,
        l.status
    FROM LOCATIONS l
    WHERE l.id = p_location_id;
END//
DELIMITER ;

--- PROCEDURE: Get picking stock levels below threshold ---
DELIMITER //
CREATE PROCEDURE get_low_picking_stock()
BEGIN
    SELECT 
        s.id,
        s.sku_code,
        s.name,
        s.min_picking_threshold,
        COALESCE(SUM(pss.current_quantity), 0) as current_picking_stock,
        (s.min_picking_threshold - COALESCE(SUM(pss.current_quantity), 0)) as deficit
    FROM SKUS s
    LEFT JOIN PICKING_STOCK_STATUS pss ON s.id = pss.sku_id
    WHERE s.min_picking_threshold > 0
    GROUP BY s.id, s.sku_code, s.name, s.min_picking_threshold
    HAVING current_picking_stock < s.min_picking_threshold
    ORDER BY deficit DESC;
END//
DELIMITER ;

--- PROCEDURE: Create reservation ---
DELIMITER //
CREATE PROCEDURE create_reservation(
    IN p_location_id INT,
    IN p_sku_id INT,
    IN p_quantity INT,
    IN p_reservation_type VARCHAR(50),
    IN p_reference_document_type VARCHAR(50),
    IN p_reference_document_id INT
)
BEGIN
    DECLARE available_capacity INT;
    
    -- Check available capacity
    SELECT (capacity - current_capacity_used - reserved_capacity)
    INTO available_capacity
    FROM LOCATIONS
    WHERE id = p_location_id;
    
    IF available_capacity >= p_quantity THEN
        INSERT INTO LOCATION_RESERVATIONS (
            location_id,
            sku_id,
            quantity,
            reservation_type,
            reference_document_type,
            reference_document_id,
            status
        ) VALUES (
            p_location_id,
            p_sku_id,
            p_quantity,
            p_reservation_type,
            p_reference_document_type,
            p_reference_document_id,
            'ACTIVE'
        );
        
        SELECT 'SUCCESS' as result, LAST_INSERT_ID() as reservation_id;
    ELSE
        SELECT 'INSUFFICIENT_CAPACITY' as result, available_capacity;
    END IF;
END//
DELIMITER ;

--- PROCEDURE: Complete reservation (convert to actual stock) ---
DELIMITER //
CREATE PROCEDURE complete_reservation(
    IN p_reservation_id INT,
    IN p_performed_by INT
)
BEGIN
    DECLARE v_location_id INT;
    DECLARE v_sku_id INT;
    DECLARE v_quantity INT;
    DECLARE v_ref_doc_type VARCHAR(50);
    DECLARE v_ref_doc_id INT;
    
    -- Get reservation details
    SELECT location_id, sku_id, quantity, reference_document_type, reference_document_id
    INTO v_location_id, v_sku_id, v_quantity, v_ref_doc_type, v_ref_doc_id
    FROM LOCATION_RESERVATIONS
    WHERE id = p_reservation_id AND status = 'ACTIVE';
    
    IF v_location_id IS NOT NULL THEN
        -- Insert into STOCK_LEDGER (this will trigger stock updates)
        -- Note: from_location_id depends on operation type
        INSERT INTO STOCK_LEDGER (
            sku_id,
            to_location_id,
            quantity,
            operation_type,
            performed_by,
            reference_document_type,
            reference_document_id
        ) VALUES (
            v_sku_id,
            v_location_id,
            v_quantity,
            'RECEIPT',  -- Adjust based on reservation_type
            p_performed_by,
            v_ref_doc_type,
            v_ref_doc_id
        );
        
        -- Mark reservation as completed
        UPDATE LOCATION_RESERVATIONS
        SET status = 'COMPLETED',
            completed_at = CURRENT_TIMESTAMP
        WHERE id = p_reservation_id;
        
        SELECT 'SUCCESS' as result;
    ELSE
        SELECT 'RESERVATION_NOT_FOUND' as result;
    END IF;
END//
DELIMITER ;

================================================================================
PART 2.5: DAILY OPERATION WORKFLOW & BON-BASED ARCHITECTURE
================================================================================

CORE CONSTRAINT:
----------------
Every operational day starts with:
1. A list of ingoing products (BON_COMMANDE)
2. A list of outgoing products (BON_PREPARATION - AI generated)
3. A FIXED number of chariots (DAILY_OPERATION.total_chariots_available)

WAREHOUSE ZONES:
----------------
The warehouse is divided into functional zones:
1. **RECEIVING**: Where products enter the warehouse
2. **VRAC** (Bulk Storage): High-capacity bulk storage for large quantities
3. **PICKING**: Intermediate storage optimized for order fulfillment
4. **EXPEDITION**: Where products leave the warehouse

CRITICAL FLOW RULE:
Products CANNOT go directly from VRAC → EXPEDITION
ENFORCED FLOW: VRAC → PICKING → EXPEDITION via REPLENISHMENT

FLOOR LAYOUT ARCHITECTURE:
---------------------------
Each warehouse has a multi-level floor structure:

**1. GROUND FLOOR (Level 0) - PICKING & OPERATIONS:**
   - Floor Type: GROUND
   - Used for: Picking operations, expedition, VRAC overflow, offices
   - Organized as a matrix of slots (rows × columns)
   - Default: 10 rows × 20 columns = 200 slots
   
   Ground Floor Sections:
   - PICKING: Main picking space for order fulfillment
   - ELEVATOR: Central elevator area
   - MONTE_CHARGE_1: Freight elevator 1
   - MONTE_CHARGE_2: Freight elevator 2
   - VRAC: Bulk storage space (overflow from main VRAC)
   - EXPEDITION_1: Expedition zone 1
   - EXPEDITION_2: Expedition zone 2
   - BUREAU: Office area

**2. STORAGE FLOORS (Levels 1-4) - PALLET STORAGE:**
   - Floor Type: STORAGE
   - Up to 4 storage floors per warehouse
   - Used for: High-density pallet storage
   - Organized as a matrix of slots (rows × columns)
   - Default: 20 rows × 30 columns = 600 slots per floor
   
   Storage Floor Sections:
   - PALETTE_SLOTS: Main pallet storage area
   - ROUTE_SLOTS: Aisles and routes between storage
   - ELEVATOR: Central elevator area
   - MONTE_CHARGE_1: Freight elevator 1
   - MONTE_CHARGE_2: Freight elevator 2

**MATRIX-BASED SLOT SYSTEM:**
Each floor is a matrix where:
- Rows and Columns define precise positions
- Slots are grouped into functional Sections
- Each Slot has a unique code (e.g., GF-R03-C12 for ground floor)
- Slots can contain:
  * Storage locations (is_storage = TRUE)
  * Infrastructure (is_infrastructure = TRUE) - elevators, offices
  * Aisles (is_aisle = TRUE) - routes for movement

**LOCATION-SLOT RELATIONSHIP:**
- Locations (storage positions) can optionally reference a Slot
- Ground floor picking locations typically reference PICKING section slots
- Storage floor locations reference PALETTE_SLOTS section slots
- VRAC locations may not have specific slot assignments (flexible)
- Slot reference enables precise matrix-based positioning

**FLOOR NAMING CONVENTIONS:**
- Ground Floor Slots: GF-R{row:02d}-C{col:02d} (e.g., GF-R01-C05)
- Storage Floor Slots: SF{level}-R{row:02d}-C{col:02d} (e.g., SF2-R10-C15)
- Locations with Slots: {warehouse_code}-{slot_code}

LOCATION STATUS STATES:
-----------------------
Each location has a status computed from capacity:
- **FREE**: current_capacity_used = 0 AND reserved_capacity = 0
- **ASSIGNED**: reserved_capacity > 0 BUT current_capacity_used = 0 (reservation pending physical arrival)
- **OCCUPIED**: current_capacity_used > 0 (physically contains stock)

CAPACITY RULE (ENFORCED BY CONSTRAINT):
current_capacity_used + reserved_capacity <= capacity

WORKFLOW SEQUENCE:
------------------

START OF DAY:
1. System creates DAILY_OPERATION record with operation_date and total_chariots_available
2. AI analyzes historical data (SKU_FORECAST) and generates BON_PREPARATION orders (forecasting)
3. AI checks picking stock levels against min_picking_threshold
4. If picking stock < threshold: AI generates BON_REPLENISHMENT
5. Supervisors review and validate/override BON_PREPARATION and BON_REPLENISHMENT orders

RECEIPT PHASE (Ingoing Products):
1. Create BON_COMMANDE linked to daily_operation_id
2. Add BON_COMMANDE_ITEMS with received quantities and reception_datetime
3. AI decides storage location (VRAC or PICKING based on is_bulk flag and demand frequency)
4. **RESERVATION PHASE**:
   a. System calls create_reservation() procedure
   b. Checks available_capacity = capacity - current_capacity_used - reserved_capacity
   c. If sufficient: Creates LOCATION_RESERVATIONS entry with status='ACTIVE'
   d. Updates LOCATIONS.reserved_capacity
   e. LOCATIONS.status changes to 'ASSIGNED'
5. Employee physically receives and places products at assigned location
6. **COMPLETION PHASE**:
   a. System calls complete_reservation() procedure
   b. Records in STOCK_LEDGER with:
      - operation_type = 'RECEIPT'
      - reference_document_type = 'BON_COMMANDE'
      - reference_document_id = bon_commande.id
      - to_location_id = assigned location
   c. Trigger updates STOCK table
   d. Converts reserved_capacity → current_capacity_used
   e. Marks reservation as 'COMPLETED'
   f. LOCATIONS.status changes to 'OCCUPIED'

TRANSFER PHASE (Storage Assignment):
1. AI calculates optimal storage location for each received SKU
2. Decision factors:
   - Distance from receiving zone
   - Weight (heavy items → lower floors)
   - Demand frequency (high demand → PICKING zone, low demand → VRAC zone)
   - Current congestion (check LOCATION_TRAFFIC)
   - Available capacity (capacity - current_capacity_used - reserved_capacity)
3. **Storage Logic**:
   - IF is_bulk = TRUE OR demand_frequency < threshold: Assign to VRAC zone
   - ELSE: Assign directly to PICKING zone
4. Supervisor can override with justification
5. Employee executes transfer
6. System records in STOCK_LEDGER with:
   - operation_type = 'TRANSFER'
   - from_location_id = receiving zone
   - to_location_id = assigned storage location

REPLENISHMENT PHASE (VRAC → PICKING):
1. **Trigger Conditions**:
   a. Automated: AI monitors PICKING_STOCK_STATUS
   b. When picking stock < min_picking_threshold: Generate BON_REPLENISHMENT
   c. Manual: Supervisor can manually trigger replenishment
2. **AI Replenishment Generation**:
   a. Query get_low_picking_stock() procedure
   b. For each SKU below threshold:
      - Find available stock in VRAC zone
      - Calculate quantity needed = min_picking_threshold - current_stock
      - Cap at max_picking_capacity
      - Select optimal VRAC location (closest, least congested)
      - Select optimal PICKING location (available capacity, proximity to expedition)
   c. Create BON_REPLENISHMENT linked to daily_operation_id
   d. Add BON_REPLENISHMENT_ITEMS with:
      - from_vrac_location_id (source VRAC location)
      - to_picking_location_id (destination PICKING location)
      - quantity
      - sequence_number (optimized route)
   e. Store decision in AI_DECISIONS with decision_type='REPLENISHMENT_GENERATION'
3. **Reservation at PICKING location**:
   a. Create LOCATION_RESERVATIONS for destination PICKING location
   b. Check capacity constraints
   c. Update reserved_capacity and status='ASSIGNED'
4. **Chariot Assignment**:
   a. Check available chariots (total_chariots_available - active_assignments)
   b. If available: Assign chariot
   c. Create CHARIOT_ASSIGNMENTS with operation_type='REPLENISHMENT'
   d. Update CHARIOTS.status = 'BUSY'
5. **Validation**:
   a. Supervisor reviews BON_REPLENISHMENT
   b. Can validate or override with justification
   c. Status: PENDING → IN_PROGRESS
6. **Execution**:
   a. Employee receives replenishment order on mobile app
   b. Navigate to VRAC location (from_vrac_location_id)
   c. Pick specified quantity
   d. Transport to PICKING location (to_picking_location_id)
   e. Place stock in picking rack
7. **Recording**:
   a. System records in STOCK_LEDGER with:
      - operation_type = 'REPLENISHMENT'
      - reference_document_type = 'BON_REPLENISHMENT'
      - from_location_id = VRAC location
      - to_location_id = PICKING location
   b. Trigger updates:
      - STOCK at both locations
      - PICKING_STOCK_STATUS.current_quantity
      - LOCATION_TRAFFIC (heatmap)
      - Converts reservation to actual stock
   c. Status: IN_PROGRESS → DONE
   d. Release chariot (CHARIOT_ASSIGNMENTS.released_at = NOW())

PREPARATION PHASE (Outgoing Planning):
1. BON_PREPARATION already created at start of day (AI forecasted from SKU_FORECAST)
2. **Picking Stock Validation**:
   a. For each BON_PREPARATION_ITEMS.quantity_to_deliver:
      - Query PICKING_STOCK_STATUS for current_picking_stock
      - IF current_picking_stock < quantity_to_deliver:
         * Trigger BON_REPLENISHMENT (see above)
         * Wait for replenishment to complete
      - ELSE: Proceed to picking
3. BON_PREPARATION_ITEMS contain:
   - delivery_id (auto-increment, serves as central Delivery ID)
   - sku_id
   - quantity_to_deliver
   - current_storage_location_id (**MUST be in PICKING zone**)
4. Status flows: AI_GENERATED → VALIDATED/OVERRIDDEN → EXECUTED

PICKING PHASE (Route Optimization):
1. System creates BON_PICKING from validated BON_PREPARATION
2. **CRITICAL VALIDATION**:
   a. Verify ALL items are in PICKING zone locations
   b. Verify sufficient quantity at each picking location
   c. **Block if any item is in VRAC** (must replenish first)
3. AI optimizes picking route:
   a. Query all picking locations from BON_PREPARATION_ITEMS.current_storage_location_id
   b. Calculate distances between all pairs using x_coord, y_coord
   c. Apply TSP algorithm (Nearest Neighbor + 2-opt):
      - Start from current chariot position or expedition zone
      - Pick nearest unpicked item
      - Repeat until all items picked
      - Return to expedition zone
   d. Considers current chariot positions (avoid collisions via CHARIOT_ASSIGNMENTS)
   e. Factors in congestion (LOCATION_TRAFFIC)
   f. Calculates total_distance
4. BON_PICKING_ITEMS include sequence_number for ordered execution
5. **Chariot Assignment**:
   a. Check available chariots
   b. Assign chariot with operation_type='PICKING'
   c. Update CHARIOTS.status = 'BUSY'
6. Employee follows optimized sequence on mobile app
7. System records in STOCK_LEDGER with:
   - operation_type = 'PICKING'
   - reference_document_type = 'BON_PICKING'
   - from_location_id = PICKING location (source)
   - to_location_id = picking rack/staging location
8. Release chariot on completion

DELIVERY PHASE (Validation):
1. Create BON_LIVRAISON linked to delivery_id
2. Employee validates delivery success/failure
3. Status: IDLE → IN_PROGRESS → DONE/FAILED
4. System records in STOCK_LEDGER with:
   - operation_type = 'DELIVERY'
   - reference_document_type = 'BON_LIVRAISON'
   - from_location_id = picking rack
   - to_location_id = EXPEDITION zone
5. Release assigned chariot

CENTRAL LINKING KEY:
--------------------
delivery_id (from BON_PREPARATION.delivery_id) links:
- BON_PREPARATION (outgoing plan)
- BON_PICKING (execution route)
- BON_LIVRAISON (delivery validation)

This enables complete traceability: delivery_id → full lifecycle tracking

CHARIOT CONSTRAINT ENFORCEMENT:
-------------------------------
1. Query DAILY_OPERATION.total_chariots_available for the day
2. Count active assignments: SELECT COUNT(*) FROM CHARIOT_ASSIGNMENTS WHERE released_at IS NULL
3. Available chariots = total_chariots_available - active_assignments
4. Operations (PICKING + REPLENISHMENT) queue if no chariots available
5. Priority: REPLENISHMENT (critical) > PICKING (scheduled)

HEATMAP IMPLEMENTATION:
-----------------------
- LOCATION_TRAFFIC: Global cumulative movement count
- DAILY_LOCATION_TRAFFIC: Per-day tracking linked to daily_operation_id
- Updated automatically via STOCK_LEDGER trigger for ALL operation types
- Used by AI for congestion-aware storage assignment and route optimization
- Visualized in supervisor dashboard for real-time monitoring
- **VRAC zone traffic** is now tracked (important for bulk movement patterns)

AI DECISION TRANSPARENCY:
-------------------------
1. AI generates decision → Store in AI_DECISIONS with confidence_score
2. Reference_id links to:
   - BON_PREPARATION.delivery_id (for PREPARATION_GENERATION)
   - BON_REPLENISHMENT.id (for REPLENISHMENT_GENERATION)
   - BON_PICKING.id (for PICKING_OPTIMIZATION)
   - Location.id (for STORAGE_ASSIGNMENT)
3. Supervisor reviews → Can validate or override
4. Override stored in AI_OVERRIDES with justification
5. Employee sees only final validated decision (abstraction layer)
6. System learns from override patterns to improve future decisions

BUSINESS RULES (APPLICATION LAYER ENFORCEMENT):
------------------------------------------------
**RULE 1**: Cannot create BON_PICKING if ANY item's current_storage_location_id is in VRAC zone
   → Must generate and complete BON_REPLENISHMENT first

**RULE 2**: Replenishment from_vrac_location_id MUST reference a location in VRAC zone
   → Validate zone_id matches VRAC zone type

**RULE 3**: Picking current_storage_location_id MUST reference a location in PICKING zone
   → Validate zone_id matches PICKING zone type

**RULE 4**: Delivery (BON_LIVRAISON) MUST originate from PICKING zone only
   → Enforce through BON_PICKING validation

**RULE 5**: VRAC zone locations cannot exceed capacity
   → Enforced by CHECK constraint: current_capacity_used + reserved_capacity <= capacity

**RULE 6**: Picking location stock cannot exceed max_picking_capacity (from SKUS table)
   → Validate before reservation: SUM(quantity) <= max_picking_capacity

**RULE 7**: Direct VRAC → EXPEDITION movement is FORBIDDEN
   → Block in application logic if from_zone=VRAC and to_zone=EXPEDITION

**RULE 8**: Reservations must be completed or cancelled before creating new ones for same (location, sku)
   → Check LOCATION_RESERVATIONS.status before new reservation

PRINCIPLE 5: TRANSACTION ATOMICITY
-----------------------------------
✓ Use database transactions for multi-step operations
✓ Rollback on any failure
✓ Validate stock availability BEFORE executing
✓ Check version conflicts for optimistic locking

Code Example:
```javascript
async function executeTransfer(sku_id, from_loc, to_loc, qty, user_id) {
  const transaction = await db.transaction();
  
  try {
    // 1. Validate stock availability
    const stock = await transaction
      .select()
      .from('stock')
      .where({sku_id, location_id: from_loc})
      .forUpdate(); // Lock row
    
    if (stock.quantity < qty) {
      throw new Error('Insufficient stock');
    }
    
    // 2. Check version (optimistic locking)
    const currentVersion = stock.version;
    
    // 3. Insert into ledger
    await transaction.insert('stock_ledger').values({
      sku_id,
      from_location_id: from_loc,
      to_location_id: to_loc,
      quantity: qty,
      operation_type: 'TRANSFER',
      performed_by: user_id
    });
    
    // 4. Update stock (via trigger or manual)
    // Trigger handles this automatically
    
    await transaction.commit();
    return { success: true };
    
  } catch (error) {
    await transaction.rollback();
    throw error;
  }
}
```

PRINCIPLE 8: AI OPTIMIZATION ALGORITHMS
----------------------------------------

Storage Optimization:
✓ Calculate distance from receipt zone to storage location
✓ Factor in: weight (heavy → lower floors), demand frequency (high → near picking)
✓ Avoid congested zones (check LOCATION_TRAFFIC)
✓ Consider slot availability

Scoring Formula Example:
```
score = (w1 * distance_score) 
      + (w2 * floor_penalty)
      + (w3 * demand_score)
      + (w4 * congestion_penalty)
      + (w5 * availability_score)

where weights (w1...w5) are stored in AI_MODEL_WEIGHTS
```

Implementation Flow:
1. Employee receives SKU at receiving zone
2. AI queries available storage locations
3. For each location, calculate:
   - Euclidean distance from receiving zone (using x_coord, y_coord)
   - Floor penalty: weight * floor_level (heavier items prefer lower floors)
   - Demand score: SKU_STATISTICS.demand_frequency (higher demand → closer to picking)
   - Congestion penalty: LOCATION_TRAFFIC.movement_count (avoid busy zones)
   - Availability: current occupied slots vs total capacity
4. Apply scoring formula with weights from AI_MODEL_WEIGHTS
5. Return top 3 locations with confidence scores
6. Store decision in AI_DECISIONS with decision_type='STORAGE_ASSIGNMENT'
7. Present to supervisor for validation or override

Picking Optimization:
✓ Shortest path algorithm (Dijkstra / A*)
✓ Traveling Salesman Problem (TSP) for multi-pick routes
✓ Sequence by distance to expedition zone
✓ Avoid chariot collisions (path planning)

Implementation Flow:
1. Input: BON_PREPARATION with items to pick
2. Query current storage locations for each SKU from BON_PREPARATION_ITEMS.current_storage_location_id
3. Build graph of all pick locations
4. Calculate distances between all pairs using coordinates
5. Apply TSP algorithm:
   a. Nearest Neighbor: Start from expedition, pick nearest unpicked item
   b. Repeat until all items picked
   c. Apply 2-opt improvement to reduce total distance
6. Assign sequence_number to each pick (1, 2, 3, ...)
7. Calculate total_distance for the route
8. Check CHARIOT_ASSIGNMENTS to avoid chariot collision in same time window
9. Create BON_PICKING with optimized route
10. Store decision in AI_DECISIONS with decision_type='PICKING_OPTIMIZATION'

Path Collision Avoidance:
```sql
SELECT ba.chariot_id, bp.id, bpi.source_storage_location_id
FROM CHARIOT_ASSIGNMENTS ba
JOIN BON_PICKING bp ON ba.bon_picking_id = bp.id
JOIN BON_PICKING_ITEMS bpi ON bp.id = bpi.bon_picking_id
WHERE ba.released_at IS NULL  -- Active assignments
AND bp.status = 'IN_PROGRESS'
```
If paths overlap, adjust sequence or delay assignment.

PART 10: AI IMPLEMENTATION GUIDELINES
================================================================================

FORECASTING MODEL:
Algorithm: Time series forecasting (ARIMA, Prophet, or LSTM)
Input Features:
  - Historical delivery quantities (FORECAST_HISTORY)
  - Day of week, seasonality
  - SKU characteristics (weight, frequency)
Output: Predicted demand per SKU
Training: Batch training weekly with FORECAST_HISTORY
Evaluation: MAE (Mean Absolute Error)

STORAGE OPTIMIZATION:
Algorithm: Multi-criteria decision making (weighted scoring)
Scoring Function:
  storage_score = 
    - w1 * euclidean_distance(receipt_zone, storage_location)
    + w2 * floor_penalty(weight, floor_level)
    + w3 * demand_frequency_bonus(sku_id)
    - w4 * congestion_penalty(location_traffic)
    + w5 * availability_score(empty_slots)

Weights (w1-w5) stored in AI_MODEL_WEIGHTS
Adaptive: Adjust weights based on override patterns

PICKING OPTIMIZATION:
Algorithm: Nearest Neighbor + 2-opt improvement (TSP)
Steps:
  1. Start from expedition zone
  2. Find nearest unpicked item
  3. Add to route
  4. Repeat until all items picked
  5. Return to expedition zone
  6. Apply 2-opt to improve route
  7. Store sequence in PICKING_ORDER_ITEMS.sequence_number

Path Collision Avoidance:
  - Check CHARIOT_ASSIGNMENTS for overlapping time windows
  - Adjust routes if paths intersect

ADAPTIVE LEARNING:
Monitor Override Patterns:
  - If location X is overridden 5+ times → decrease its weight
  - If supervisor prefers floor Y → increase its weight
Update Frequency: Weekly batch updates
Metrics to Track:
  - Override rate per decision type (from AI_OVERRIDES)
  - Forecast accuracy (MAE from SKU_FORECAST)
  - Route efficiency (total_distance: AI vs override)

Learning Algorithm:
```javascript
async function adaptWeights() {
  // 1. Query overrides from last week
  const overrides = await db
    .select('ai_decision_id', 'decision_type', 'new_decision')
    .from('AI_OVERRIDES')
    .where('created_at', '>=', oneWeekAgo);

  // 2. Analyze patterns
  const locationOverrides = overrides
    .filter(o => o.decision_type === 'STORAGE_ASSIGNMENT')
    .map(o => JSON.parse(o.new_decision).location_id);
  
  const overrideFrequency = countFrequency(locationOverrides);

  // 3. Adjust weights
  for (const [locationId, count] of overrideFrequency) {
    if (count >= 5) {
      // Decrease weight for AI-rejected locations
      await decreaseWeight('congestion_penalty', 0.05);
      await increaseWeight('demand_score', 0.05);
    }
  }

  // 4. Evaluate forecast accuracy
  const forecasts = await db
    .select('predicted_quantity', 'actual_quantity')
    .from('SKU_FORECAST')
    .where('actual_quantity', 'IS NOT', null);
  
  const mae = forecasts.reduce((sum, f) => 
    sum + Math.abs(f.predicted_quantity - f.actual_quantity), 0
  ) / forecasts.length;

  // 5. Store performance metric
  await db.insert('AI_PERFORMANCE_METRICS').values({
    model_type: 'FORECAST',
    metric_name: 'MAE',
    value: mae
  });
}
```
